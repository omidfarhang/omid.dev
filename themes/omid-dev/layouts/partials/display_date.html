{{- /* Locale-aware visible date; ISO in title/datetime attributes. Params (dict): date — time.Time (required) lang — language code (required) style — "short" | "medium" | "long" (default "long"; non-fa only) tag — "span" | "time" | "none" (default "span") class — optional class for span/time title — optional title override (defaults to ISO 8601) */ -}} {{- $date := .date -}} {{- $lang := .lang -}} {{- $style := .style | default "long" -}} {{- $tag := .tag | default "span" -}} {{- $iso := $date.Format "2006-01-02T15:04:05Z07:00" -}} {{- $title := .title | default $iso -}} {{- $text := "" -}} {{- if eq $lang "fa" -}} {{- $text = partial "jalali_date.html" $date -}} {{- else if eq $style "short" -}} {{- $text = $date.Format "Jan 2, 2006" -}} {{- else if eq $style "medium" -}} {{- $text = $date | time.Format ":date_medium" -}} {{- else -}} {{- $text = $date | time.Format (default "January 2, 2006" site.Params.DateFormat) -}} {{- end -}} {{- if eq $tag "none" -}} {{- $text -}} {{- else if eq $tag "time" -}} {{ $text }} {{- else -}} {{ $text }} {{- end -}}
